cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(CronusInterceptor VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Visual Studio 2022, Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX GR_STATIC)
add_compile_options(/W4 /WX /utf-8 /permissive- /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /PDBALTPATH:%_PDB%)
option(CRONUS_ASAN "Build validation with AddressSanitizer" OFF)
if(CRONUS_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/DEBUG /INCREMENTAL:NO)
endif()
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
 set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/lib")
endforeach()
add_library(InterceptorSDK STATIC Interceptor/Interceptor.cpp)
target_include_directories(InterceptorSDK PUBLIC Interceptor)
target_link_libraries(InterceptorSDK PUBLIC setupapi newdev cfgmgr32 hid advapi32 user32 comctl32 ole32)
add_library(GpcCore STATIC App/Core/Gpc.cpp App/Core/Persistence.cpp App/Core/ControlFrame.cpp App/Core/ScriptCompiler.cpp App/Core/ScriptInstance.cpp)
target_link_libraries(GpcCore PUBLIC bcrypt advapi32)
add_library(CaptureEngine STATIC App/Capture/Remapper.cpp App/Capture/Service.cpp App/Capture/DriverMonitor.cpp)
target_link_libraries(CaptureEngine PUBLIC GpcCore InterceptorSDK psapi)
add_library(StudioUI STATIC App/UI/Window.cpp)
target_link_libraries(StudioUI PUBLIC CaptureEngine comdlg32 comctl32 gdi32 shell32 ole32 windowscodecs uuid)
add_executable(CronusInterceptor WIN32 App/Main.cpp App/App.rc)
target_link_libraries(CronusInterceptor PRIVATE StudioUI)
target_link_options(CronusInterceptor PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_SOURCE_DIR}/App/App.manifest")
add_custom_command(TARGET CronusInterceptor POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/Scripts" "$<TARGET_FILE_DIR:CronusInterceptor>/Scripts")
add_executable(GpcTests App/Tests/GpcTests.cpp)
target_link_libraries(GpcTests PRIVATE GpcCore)
enable_testing()
add_test(NAME GpcSemantics COMMAND GpcTests "${CMAKE_SOURCE_DIR}/Scripts")
add_executable(SemanticTests App/Tests/Semantics.cpp)
target_link_libraries(SemanticTests PRIVATE GpcCore InterceptorSDK)
add_test(NAME GpcConformanceAndStress COMMAND SemanticTests)
add_executable(PersistenceTests App/Tests/PersistenceTests.cpp)
target_link_libraries(PersistenceTests PRIVATE GpcCore)
add_test(NAME IsolatedPersistentMemory COMMAND PersistenceTests)
